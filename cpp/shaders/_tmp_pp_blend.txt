vec2 glassBlueNoiseR2(uint index)
{
    const float g = 1.32471795724474602596;
    return fract(vec2(float(index) / g, float(index) / (g * g)));
}

float glassBlueNoiseSample(uvec2 pixelIdx, uint temporalIdx, uint seedIndex)
{
    const uvec2 seedOffset = uvec2(glassBlueNoiseR2(seedIndex) * vec2(128.0));
    const uint x = (pixelIdx.x + seedOffset.x) & 127u;
    const uint y = (pixelIdx.y + seedOffset.y) & 127u;
    const uint layer = temporalIdx & 63u;
    const uint byteIndex = 16384u * layer + 128u * y + x;
    const uint word = glassBlueNoise.words[byteIndex >> 2u];
    const uint raw = (word >> ( (byteIndex & 3u) * 8u)) & 0xFFu;
    return min(float(raw) * (1.0 / 255.0), 1.0 - 1.0 / 255.0);
}

vec3 glassAlphaBlendDecodeNormal(vec2 encoded)
{
    vec3 v = vec3(encoded.x, encoded.y, 1.0 - abs(encoded.x) - abs(encoded.y));
    const float t = clamp(- v.z, 0.0, 1.0);
    v.xy += vec2(v.x >= 0.0 ? - t : t, v.y >= 0.0 ? - t : t);
    return normalize(v);
}

uint glassAlphaBlendHash(uint state)
{
    state ^= state >> 16u;
    state *= 0x7FEB352Du;
    state ^= state >> 15u;
    state *= 0x846CA68Bu;
    state ^= state >> 16u;
    return state;
}

float glassAlphaBlendRandom(inout uint state)
{
    state = glassAlphaBlendHash(state + 0x9E3779B9u);
    return float(state & 0x00FFFFFFu) * (1.0 / 16777216.0);
}

bool glassAlphaBlendInside(ivec2 pixel, ivec2 extent)
{
    return pixel.x >= 0 && pixel.y >= 0 &&
           pixel.x < extent.x && pixel.y < extent.y;
}



float glassAlphaBlendSceneDistance(vec2 ndc, float depthValue)
{
    if (depthValue <= 1.0e-5)
    {
        return GLASS_ALPHA_BLEND_NO_GEOMETRY_DISTANCE;
    }

    vec4 sceneViewPosition = cam.projInverse * vec4(ndc, depthValue, 1.0);
    if (abs(sceneViewPosition.w) <= 1.0e-6)
    {
        return GLASS_ALPHA_BLEND_NO_GEOMETRY_DISTANCE;
    }

    return length(sceneViewPosition.xyz / sceneViewPosition.w);
}

vec3 glassAlphaBlendPixelWorldDirection(vec2 ndc)
{
    vec4 viewPosition = cam.projInverse * vec4(ndc, 1.0, 1.0);
    vec3 viewDirection = normalize(viewPosition.xyz / max(abs(viewPosition.w), 1.0e-6));
    return normalize( (cam.viewInverse * vec4(viewDirection, 0.0)).xyz);
}






vec3 glassAlphaBlendNeighborLight(ivec2 neighbor)
{
    vec3 radiance = max(imageLoad(sceneColor, neighbor).rgb, vec3(0.0));
    vec3 albedo = clamp(imageLoad(sceneDiffuseAlbedo, neighbor).rgb, vec3(0.0), vec3(1.0));



    if (dot(albedo, GLASS_ALPHA_BLEND_LUMA) < GLASS_ALPHA_BLEND_ALBEDO_FLOOR)
    {
        return radiance;
    }
    return radiance / max(albedo, vec3(GLASS_ALPHA_BLEND_ALBEDO_FLOOR));
}

struct SphericalHarmonic
{
    vec4 shY;
    vec2 cocg;
};



vec3 shRgbToYCoCg(vec3 color)
{
    return vec3(
        0.25 * color.r + 0.5 * color.g + 0.25 * color.b,
        0.5 * color.r + 0.0 * color.g - 0.5 * color.b,
        - 0.25 * color.r + 0.5 * color.g - 0.25 * color.b);
}






vec3 shYCoCgToRgb(vec3 color)
{
    return vec3(
        color.x + color.y - color.z,
        color.x + color.z,
        color.x - color.y - color.z);
}


SphericalHarmonic shCreate()
{
    SphericalHarmonic sh;
    sh.shY = vec4(0.0);
    sh.cocg = vec2(0.0);
    return sh;
}


vec4 shEvalBasis(vec3 D)
{
    const float L00 = (0.5 * sqrt(1.0 / 3.14159265359));
    const float L1_1 = (0.5 * sqrt(3.0 / 3.14159265359)) * D.y;
    const float L10 = (0.5 * sqrt(3.0 / 3.14159265359)) * D.z;
    const float L11 = (0.5 * sqrt(3.0 / 3.14159265359)) * D.x;
    return vec4(L11, L1_1, L10, L00);
}


void shAddRadiance(inout SphericalHarmonic sh, vec3 color, vec3 dir)
{
    const vec3 YCoCg = shRgbToYCoCg(color);
    const float Y = YCoCg.x;


    if (any(isnan(dir)))
    {
        dir = vec3(0.0);
    }

    sh.cocg += YCoCg.yz;
    sh.shY += shEvalBasis(dir) * Y;
}


vec3 shGetIrradiance(SphericalHarmonic sh)
{
    const float Y = max(0.0, sh.shY.w);
    return shYCoCgToRgb(vec3(Y, sh.cocg * (0.5 * sqrt(1.0 / 3.14159265359))));
}


vec3 shResolveColorToDiffuse(SphericalHarmonic sh, vec3 N, float cosHalfAngle)
{
    const float d = dot(sh.shY.xyz, N);
    float Y = 0.25 * sqrt(4.0 * 3.14159265359) * sh.shY.w +
              0.5 * sqrt(4.0 * 3.14159265359 / 3.0) * d;


    Y *= (2.0 * 3.14159265359 * (1.0 - cosHalfAngle)) / 3.14159265359;


    const float modifier =
        (sh.shY.w == 0.0 || Y == 0.0) ? 0.0 : ( (0.5 * sqrt(1.0 / 3.14159265359)) * Y / sh.shY.w);
    const vec2 adjCoCg = sh.cocg * modifier;

    return shYCoCgToRgb(vec3(Y, adjCoCg));
}


void shScale(inout SphericalHarmonic sh, float scalar)
{
    sh.shY *= scalar;
    sh.cocg *= scalar;
}


void shHanningFilter(inout SphericalHarmonic sh, float k)
{
    sh.shY.xyz *= (1.0 + cos(3.14159265359 / k)) * 0.5;
}


vec3 shEvaluateHenyeyGreenstein(SphericalHarmonic sh, vec3 D, float g)
{
    const vec4 shBasis = shEvalBasis(D);
    const vec4 hgBasis = vec4(vec3(g), 1.0);
    const vec4 hgProjectedZH = hgBasis * shBasis;
    const float resultY = dot(sh.shY, hgProjectedZH);

    const float modifier =
        (sh.shY.w == 0.0 || resultY == 0.0) ? 0.0 : ( (0.5 * sqrt(1.0 / 3.14159265359)) * resultY / sh.shY.w);
    const vec2 adjCoCg = sh.cocg * modifier;

    return max(vec3(0.0), shYCoCgToRgb(vec3(resultY, adjCoCg)));
}


SphericalHarmonic shLerp(SphericalHarmonic x, SphericalHarmonic y, float s)
{
    SphericalHarmonic result;
    result.shY = mix(x.shY, y.shY, s);
    result.cocg = mix(x.cocg, y.cocg, s);
    return result;
}

vec3 glassAlphaBlendSampleVolumeSh(vec3 worldPosition)
{
    if (pc.shValid < 0.5 || pc.shFroxelW == 0u || pc.shFroxelH == 0u || pc.shFroxelDepth == 0u)
    {
        return vec3(0.0);
    }


    const mat4 viewMatrix = inverse(cam.viewInverse);
    const mat4 projMatrix = inverse(cam.projInverse);
    const vec4 viewPosition = viewMatrix * vec4(worldPosition, 1.0);
    const vec4 clipPosition = projMatrix * viewPosition;

    if (abs(clipPosition.w) < 1.0e-6)
    {
        return vec3(0.0);
    }

    const vec2 ndc = clipPosition.xy / clipPosition.w;

    const vec2 uv = ndc * vec2(0.5, - 0.5) + 0.5;

    if (any(lessThan(uv, vec2(0.0))) || any(greaterThan(uv, vec2(1.0))))
    {
        return vec3(0.0);
    }

    const float viewZ = - viewPosition.z;
    const float nearPlane = max(cam.cameraParams.x, 1.0e-3);
    const float t = clamp( (viewZ - nearPlane) / max(pc.shMaxDistance, 1.0), 0.0, 1.0);

    const float sliceT = pow(t, 1.0 / max(pc.shDepthExponent, 1.0));

    const ivec3 size = ivec3(int(pc.shFroxelW), int(pc.shFroxelH), int(pc.shFroxelDepth));
    const vec3 coordinate = vec3(uv * vec2(size.xy), sliceT * float(size.z));

    const vec3 clamped = clamp(coordinate, vec3(0.0), vec3(size - ivec3(1)));
    const ivec3 base = ivec3(floor(clamped));
    const ivec3 upper = min(base + ivec3(1), size - ivec3(1));
    const vec3 fraction = clamped - vec3(base);

    SphericalHarmonic sh = shCreate();

    for (int dz = 0; dz < 2; ++ dz)
    {
        for (int dy = 0; dy < 2; ++ dy)
        {
            for (int dx = 0; dx < 2; ++ dx)
            {
                const ivec3 p = ivec3(
                    dx == 0 ? base.x : upper.x,
                    dy == 0 ? base.y : upper.y,
                    dz == 0 ? base.z : upper.z);

                const float w =
                    (dx == 0 ? 1.0 - fraction.x : fraction.x) *
                    (dy == 0 ? 1.0 - fraction.y : fraction.y) *
                    (dz == 0 ? 1.0 - fraction.z : fraction.z);

                sh.shY += imageLoad(volumeShY, p) * w;
                sh.cocg += imageLoad(volumeShCoCg, p).xy * w;
            }
        }
    }


    shHanningFilter(sh, 0.7);

    return shGetIrradiance(sh);
}
